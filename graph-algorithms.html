图算法

Graph Algorithms

📖什么是图算法？

图算法是在图数据结构上执行的各种算法，用于解决路径查找、连通性检测、最短路径等问题。图由顶点(V)和边(E)组成，可以表示实体之间的关系。

💡 核心直觉
想象一张城市地图，城市是顶点，道路是边。图算法就是帮你找到从一个城市到另一个城市的最短路径，或者判断两个城市是否连通。

📐核心算法

🔍 遍历算法
BFS（广度优先搜索）O(V+E)
$$T = O(|V| + |E|)$$
逐层遍历，用于最短路径（无权图）
JavaScript 
// BFS 广度优先搜索
function bfs(graph, start) {
    const queue = [start];
    const visited = new Set([start]);
    const order = [];

    while (queue.length > 0) {
        const current = queue.shift();
        order.push(current);

        // 遍历相邻节点
        graph[current].forEach(neighbor => {
            if (!visited.has(neighbor)) {
                visited.add(neighbor);
                queue.push(neighbor);
            }
        });
    }
    return order;
}
DFS（深度优先搜索）O(V+E)
$$T = O(|V| + |E|)$$
深入到底再回溯，用于连通性检测
JavaScript 
// DFS 深度优先搜索 (迭代版)
function dfs(graph, start) {
    const stack = [start];
    const visited = new Set();
    const order = [];

    while (stack.length > 0) {
        const current = stack.pop();
        
        if (!visited.has(current)) {
            visited.add(current);
            order.push(current);

            // 逆序入栈以保持顺序
            [...graph[current]].reverse().forEach(neighbor => {
                if (!visited.has(neighbor)) {
                    stack.push(neighbor);
                }
            });
        }
    }
    return order;
}
📍 最短路径
Dijkstra算法O((V+E)logV)
$$d[v] = \min(d[v], d[u] + w(u,v))$$
单源最短路径，边权非负
JavaScript 
// Dijkstra 最短路径算法
function dijkstra(graph, start) {
    const dist = {};
    const visited = new Set();
    
    // 初始化距离
    for (let node in graph) {
        dist[node] = node === start ? 0 : Infinity;
    }

    while (visited.size < Object.keys(graph).length) {
        // 找到未访问的最小距离节点
        let minNode = null;
        let minDist = Infinity;
        
        for (let node in dist) {
            if (!visited.has(node) && dist[node] < minDist) {
                minDist = dist[node];
                minNode = node;
            }
        }

        if (minNode === null) break;
        visited.add(minNode);

        // 松弛操作
        for (let neighbor in graph[minNode]) {
            const weight = graph[minNode][neighbor];
            const newDist = dist[minNode] + weight;
            
            if (newDist < dist[neighbor]) {
                dist[neighbor] = newDist;
            }
        }
    }
    return dist;
}
