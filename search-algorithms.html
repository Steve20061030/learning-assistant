Search Algorithms

📖 什么是查找算法？

查找算法是在数据集合中寻找特定元素的算法。查找操作是计算机科学中最基础也最常用的操作之一。

常见的查找算法包括顺序查找、二分查找、哈希查找、二叉搜索树查找等。

💡 核心直觉
想象你在图书馆找一本书。顺序查找就像从第一排开始一本一本地找；二分查找就像先看目录，然后直接找到对应的书架。

📊 算法复杂度

⚡ 二分查找
二分查找 O(log n)
$$T(n) = T(n/2) + O(1)$$
每次将搜索范围减半，需要有序数组
🔄 顺序查找
线性查找 O(n)
$$T(n) = \sum_{i=1}^{n} 1 = O(n)$$
逐个检查每个元素
🔍 哈希查找
哈希表查找 O(1)
$$T(n) = O(1)$$
平均情况，需要处理哈希冲突

🔍 二分查找 (Binary Search)
JavaScript - 经典实现 
// 二分查找 - 经典实现
function binarySearch(arr, target) {
    let left = 0;
    let right = arr.length - 1;
    
    while (left <= right) {
        const mid = Math.floor((left + right) / 2);
        
        if (arr[mid] === target) {
            return mid;
        } else if (arr[mid] < target) {
            left = mid + 1;
        } else {
            right = mid - 1;
        }
    }
    return -1;
}

// 二分查找 - 左边界
function lowerBound(arr, target) {
    let left = 0, right = arr.length;
    
    while (left < right) {
        const mid = floor((left + right) / 2);
        if (arr[mid] < target) {
            left = mid + 1;
        } else {
            right = mid;
        }
    }
    return left;
}

// 示例
const arr = [1, 3, 5, 7, 9, 11];
console.log(binarySearch(arr, 7)); // 3
🌳 二叉搜索树 (BST)
JavaScript - BST实现 
// 二叉搜索树节点
class TreeNode {
    constructor(val) {
        this.val = val;
        this.left = null;
        this.right = null;
    }
}

// BST插入
function bstInsert(root, val) {
    if (!root) return new TreeNode(val);
    
    if (val < root.val) {
        root.left = bstInsert(root.left, val);
    } else {
        root.right = bstInsert(root.right, val);
    }
    return root;
}

// BST搜索 - O(log n)
function bstSearch(root, target) {
    if (!root || root.val === target) return root;
    
    if (target < root.val) {
        return bstSearch(root.left, target);
    }
    return bstSearch(root.right, target);
}

// BST中序遍历 (有序)
function inorderTraversal(root, result = []) {
    if (root) {
        inorderTraversal(root.left, result);
        result.push(root.val);
        inorderTraversal(root.right, result);
    }
    return result;
}
🔑 哈希查找 (Hash Search)
JavaScript - 哈希表 
// 简单哈希表实现
class HashTable {
    constructor(size = 100) {
        this.size = size;
        this.table = new Array(size).fill(null);
    }
    
    hash(key) {
        let hash = 0;
        for (let char of key) {
            hash = (hash * 31 + char.charCodeAt(0)) % this.size;
        }
        return hash;
    }
    
    set(key, value) {
        const index = this.hash(key);
        this.table[index] = value;
    }
    
    get(key) {
        const index = this.hash(key);
        return this.table[index];
    }
}

// 使用示例 - O(1) 平均查找
const ht = new HashTable();
ht.set("apple", 100);
ht.set("banana", 200);
console.log(ht.get("apple")); // 100
